# STM32云台通信调试指南

## 概述

本文档为STM32云台控制系统的通信调试提供详细的指导，包括常见问题诊断、性能优化和最佳实践。

## 快速问题解决

### 🚨 连接频繁断开

**症状**：云台运行一段时间后进入锁定保护模式；在当前测试板上会表现为锁定指示状态

**可能原因**：
1. 上位机握手后未持续发送心跳包
2. `search` 或 `auto` 保活周期过长
3. USB传输延迟抖动或主机负载过高

**解决方案**：
```powershell
# 使用当前协议对应的推荐默认值
python tools\usb_host.py --port COM6 --workflow auto --heartbeat-period-ms 200 --search-period-ms 200 --auto-period-ms 5
```

当前固件侧超时阈值保持不变：

- `SEARCH` 连续 `500ms` 未收到通讯包后回退到 `STABLE`
- `AUTO_AIM` 连续 `200ms` 未收到通讯包后回退到 `SEARCH`
- 心跳包中断超过 `1000ms` 进入 `LOCK_PROTECT`

### 🚨 数据包CRC错误

**症状**：接收到数据但CRC校验失败

**可能原因**：
1. USB线缆质量问题
2. 电磁干扰
3. 缓冲区溢出

**解决方案**：

- 先核对帧头 `AA 55`、帧尾 `5A A5`、长度字段和 CRC 计算范围
- 用 `tools/usb_protocol.py` 里的 `FrameParser` 做离线复核
- 当前协议没有重传指令，CRC 错误会直接丢弃，不会自动请求重发

### 🚨 自瞄模式频繁超时

**症状**：进入自瞄模式后很快回退到搜索模式

**可能原因**：
1. `0x84` 自瞄包发送间隔超过 `200ms`
2. 视觉算法处理时间过长
3. USB主机线程阻塞，导致周期包发不出去

**解决方案**：

- 保持 `0x84` 的发送周期明显小于 `200ms`，当前 CLI 默认是 `5ms`
- 如果使用 CLI，优先用 `--workflow auto`，再按需切换到 `auto <yaw_deg> <pitch_deg>`
- 视觉侧不要在单线程里串行处理采集、识别和串口发送

### 实时性能监控

```python
class PerformanceMonitor:
    def __init__(self):
        self.metrics = {
            'tx_count': 0,
            'rx_count': 0,
            'avg_latency': 0.0,
            'max_latency': 0.0,
            'packet_loss': 0.0,
            'connection_uptime': 0.0
        }
        self.latency_history = deque(maxlen=1000)

    def log_transmission(self, timestamp):
        self.metrics['tx_count'] += 1

    def log_reception(self, timestamp, original_timestamp):
        self.metrics['rx_count'] += 1

        latency = timestamp - original_timestamp
        self.latency_history.append(latency)

        # 更新延迟统计
        self.metrics['avg_latency'] = sum(self.latency_history) / len(self.latency_history)
        self.metrics['max_latency'] = max(self.latency_history)

        # 计算丢包率
        if self.metrics['tx_count'] > 0:
            self.metrics['packet_loss'] = (1 - self.metrics['rx_count'] / self.metrics['tx_count']) * 100

    def generate_report(self):
        return f"""
通信性能报告：
- 发送包数: {self.metrics['tx_count']}
- 接收包数: {self.metrics['rx_count']}
- 平均延迟: {self.metrics['avg_latency']:.2f}ms
- 最大延迟: {self.metrics['max_latency']:.2f}ms
- 丢包率: {self.metrics['packet_loss']:.2f}%
- 连接时长: {self.metrics['connection_uptime']:.1f}s
"""
```

void UpdateSystemMonitor(SystemMonitor_t *monitor) {
    // FreeRTOS堆监控
    monitor->free_heap = xPortGetFreeHeapSize();
    monitor->min_free_heap = xPortGetMinimumEverFreeHeapSize();

    // 栈使用监控
    monitor->max_stack_usage = uxTaskGetStackHighWaterMark(NULL);

    // CPU使用率估算
    static uint32_t last_idle_count = 0;
    uint32_t current_idle_count = ulTaskGetIdleRunTimeCounter();
    uint32_t idle_delta = current_idle_count - last_idle_count;

    // 简化的CPU使用率计算
    monitor->cpu_usage_percent = 100 - (idle_delta * 100 / configTICK_RATE_HZ);
    last_idle_count = current_idle_count;

### 1. 协议分析器

```python
class ProtocolAnalyzer:
    def __init__(self):
        self.packet_log = []

    def analyze_packet(self, data: bytes, direction: str):
        try:
            # 解析协议头
            if len(data) >= 8:
                sof = data[0:2]
                length = data[2]
                cmd = data[3]

                analysis = {
                    'timestamp': time.time(),
                    'direction': direction,  # 'TX' or 'RX'
                    'valid_sof': sof == b'\xAA\x55',
                    'length': length,
                    'command': f'0x{cmd:02X}',
                    'total_length': len(data),
                    'raw_data': data.hex()
                }

                # CRC校验
                if len(data) >= length + 8:
                    expected_crc = struct.unpack('<H', data[4+length:6+length])[0]
                    actual_crc = crc16_modbus(data[:4+length])
                    analysis['crc_valid'] = expected_crc == actual_crc
                    analysis['crc_expected'] = expected_crc
                    analysis['crc_actual'] = actual_crc

                self.packet_log.append(analysis)

        except Exception as e:
            print(f"Protocol analysis error: {e}")

    def print_statistics(self):
        if not self.packet_log:
            return

        total_packets = len(self.packet_log)
        tx_packets = sum(1 for p in self.packet_log if p['direction'] == 'TX')
        rx_packets = sum(1 for p in self.packet_log if p['direction'] == 'RX')
        crc_errors = sum(1 for p in self.packet_log if not p.get('crc_valid', True))

        print(f"""
协议统计：
- 总包数: {total_packets}
- 发送包: {tx_packets}
- 接收包: {rx_packets}
- CRC错误: {crc_errors} ({crc_errors/total_packets*100:.1f}%)
""")

        # 命令统计
        cmd_stats = {}
        for packet in self.packet_log:
            cmd = packet['command']
            cmd_stats[cmd] = cmd_stats.get(cmd, 0) + 1

        print("命令分布:")
        for cmd, count in sorted(cmd_stats.items()):
            print(f"  {cmd}: {count}次")
```

### 2. 延迟测量

```c
// STM32端高精度时间戳
static uint64_t GetHighResolutionTimestamp(void) {
    // 使用DWT计数器获得微秒级精度
    static uint8_t initialized = 0;

    if (!initialized) {
        // 启用DWT
        CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;
        DWT->CTRL |= DWT_CTRL_CYCCNTENA_Msk;
        initialized = 1;
    }

    uint32_t cycles = DWT->CYCCNT;
    return cycles / (SystemCoreClock / 1000000);  // 转换为微秒
}

void USB_SendTimedFeedback(uint32_t now_ms) {
    uint64_t precise_timestamp = GetHighResolutionTimestamp();

    // 在数据包中包含高精度时间戳用于延迟测量
    uint8_t payload[USB_APP_FEEDBACK_LEN + 8];

    // 原始反馈数据
    USB_WriteU16Le(&payload[0], (uint16_t)yaw);
    USB_WriteU16Le(&payload[2], (uint16_t)pitch);
    USB_WriteU16Le(&payload[4], 0U);
    USB_WriteU32Le(&payload[6], now_ms);
    payload[10] = (uint8_t)g_usb.mode;
    payload[11] = 0U;

    // 高精度时间戳
    USB_WriteU32Le(&payload[12], (uint32_t)(precise_timestamp & 0xFFFFFFFF));
    USB_WriteU32Le(&payload[16], (uint32_t)(precise_timestamp >> 32));

    USB_SendFrame(USB_CMD_GIMBAL_FEEDBACK_PRECISE, payload, USB_APP_FEEDBACK_LEN + 8);
}
```

### 3. 自动化测试

```python
class AutomatedTester:
    def __init__(self, session: UsbSession):
        self.session = session
        self.test_results = []

    def run_stability_test(self, duration_hours: float):
        """运行长期稳定性测试"""
        print(f"开始{duration_hours}小时稳定性测试...")

        start_time = time.time()
        end_time = start_time + duration_hours * 3600

        connection_events = []

        try:
            while time.time() < end_time:
                # 检查连接状态
                if not self.session.connected:
                    event = {
                        'type': 'disconnect',
                        'time': time.time() - start_time,
                        'duration': time.time() - start_time
                    }
                    connection_events.append(event)

                    # 尝试重连
                    self.session.connect(self.session.port)

                # 模拟不同的操作模式
                mode = int((time.time() - start_time) % 60) // 20
                if mode == 0:
                    self.session.start_search()
                elif mode == 1:
                    self.session.start_auto(10.0, 5.0)
                else:
                    self.session.start_auto(-10.0, -5.0)

                time.sleep(0.1)

        except KeyboardInterrupt:
            print("测试被用户中断")

        # 生成测试报告
        total_disconnections = len(connection_events)
        print(f"""
稳定性测试报告：
- 测试时长: {(time.time() - start_time)/3600:.1f}小时
- 断连次数: {total_disconnections}
- 平均连接时长: {(time.time() - start_time)/max(total_disconnections, 1)/60:.1f}分钟
""")

        return {
            'duration_hours': (time.time() - start_time) / 3600,
            'disconnections': total_disconnections,
            'events': connection_events
        }

    def run_latency_test(self, samples: int = 1000):
        """运行延迟测试"""
        print(f"开始{samples}次延迟测试...")

        latencies = []

        for i in range(samples):
            start_time = time.time()

            # 发送控制指令
            self.session.start_auto(0.0, 0.0, periodic=False)

            # 等待反馈
            try:
                event = self.session.get_event(timeout=0.1)
                if event.kind == 'feedback':
                    latency = (time.time() - start_time) * 1000  # 转换为毫秒
                    latencies.append(latency)

            except queue.Empty:
                print(f"样本{i+1}: 超时")

            time.sleep(0.01)  # 100Hz测试频率

        if latencies:
            avg_latency = sum(latencies) / len(latencies)
            max_latency = max(latencies)
            min_latency = min(latencies)

            print(f"""
延迟测试报告：
- 有效样本: {len(latencies)}/{samples}
- 平均延迟: {avg_latency:.2f}ms
- 最大延迟: {max_latency:.2f}ms
- 最小延迟: {min_latency:.2f}ms
- 成功率: {len(latencies)/samples*100:.1f}%
""")

        return latencies
```
